Fill with voice

Use case: you want microphone capture to write into the same FieldStore used by your generated or bound form surface. The hosted bridge stays at ../voice-bridge.html.

Prerequisites

  • A published page already loaded into the client store.
  • A working exchange token with the same allowlisted origin as the host app.
  • Permission to capture microphone input in the browser or native WebView.

JavaScript

import { Comprehendly, mountVoiceBridge } from '../src/index.js'

const client = new Comprehendly({ publishableKey: 'pk_test_…' })
let tokenExpiresAt = 0

async function ensureExchangeToken() {
  if (Date.now() < tokenExpiresAt - 60_000) return false
  const session = await client.exchange()
  tokenExpiresAt = Date.now() + (session.expires_in * 1000)
  return true
}

function mountVoice() {
  mountVoiceBridge('#voice', {
    token: client.accessToken,
    pageId: '00000000-0000-0000-0000-000000000000',
    mode: 'assistant',
    store: client.store
  })
}

await ensureExchangeToken()
await client.page.get('00000000-0000-0000-0000-000000000000')
mountVoice()

window.addEventListener('message', (event) => {
  const voiceFrame = document.querySelector('#voice iframe')
  if (!voiceFrame) return

  const bridgeURL = new URL(voiceFrame.src ?? '../voice-bridge.html', window.location.href)
  const embedOrigin = new URL(bridgeURL.searchParams.get('base_url') ?? 'https://sdk.comprehendly.nz').origin
  if (event.source !== voiceFrame.contentWindow) return
  if (![bridgeURL.origin, embedOrigin].includes(event.origin)) return

  const fieldValues = event.data?.field_values ?? event.data?.fieldValues
  if (event.data?.type === 'stepcare:fields_updated' && fieldValues) {
    console.log('Field values updated from voice', fieldValues)
  }
})

Swift

import ComprehendlyForms

let client = ComprehendlyClient(publishableKey: "pk_test_…")
try await client.exchange()
_ = try await client.pageGet("00000000-0000-0000-0000-000000000000")

let voiceURL = try client.voiceBridgeURL(
  pageId: "00000000-0000-0000-0000-000000000000",
  mode: "assistant"
)

VoiceBridgeView(url: voiceURL, store: client.store)

If the short-lived token is renewed later, remount the JavaScript bridge or recreate the Swift bridge URL so the hosted bridge uses the new token. In this hosted-bridge flow, listen for the forwarded message from the mounted voice-bridge.html iframe rather than trusting unrelated frames from the same embed origin.

Limits

  • Do not reimplement the Realtime client in v1. Keep voice on the hosted voice-bridge.html / forms fill embed.
  • Forms embed v1.3 posts stepcare:fields_updated; use that event to hydrate host UI if needed.
  • Voice updates the shared store, but save still happens through forms.submissions.save with field_name-keyed values.
