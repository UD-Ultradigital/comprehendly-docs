Render a form

Use case: your app does not already have a form UI and you want the SDK to render a simple, generated surface from a published page.

Prerequisites

  • A published page_id.
  • A publishable key with an allowlisted client origin.
  • forms.schemas:read and forms.submissions:write on the exchanged token.

JavaScript

import { Comprehendly } from '../src/index.js'

const client = new Comprehendly({
  publishableKey: 'pk_test_…'
})

await client.exchange()
const page = await client.page.get('00000000-0000-0000-0000-000000000000')

const form = document.getElementById('generated-form')
form.innerHTML = ''

for (const field of client.store.fields()) {
  if (!field.generated) continue

  const label = document.createElement('label')
  label.textContent = field.label || field.fieldName

  const input = document.createElement('input')
  input.addEventListener('input', () => {
    client.store.set({
      fieldName: field.fieldName,
      value: input.value,
      source: 'user'
    })
  })

  form.append(label, input)
}

Swift

import ComprehendlyForms
import SwiftUI

struct GeneratedFormScreen: View {
  @StateObject private var client = ComprehendlyClient(publishableKey: "pk_test_…")
  let pageId = "00000000-0000-0000-0000-000000000000"

  var body: some View {
    ComprehendlyFormView(store: client.store)
      .task {
        try await client.exchange()
        _ = try await client.pageGet(pageId)
      }
  }
}

Limits

  • Generated UI is a linear list of primitive controls, not a clone of editor layout, grids, or media.
  • form_field_rating and form_field_score are loaded into the store but not rendered by the v1 generated surface.
  • If you already have branded or custom controls, switch to Bind existing fields instead of layering both surfaces together.
